def site_request_id(n):
    """vLLM v0.30's request id form: "chatcmpl-" + 36-char uuid + "-" + 8 characters (same as t51-glm53-p2d's
    record-bytes measurement)."""
    return f"chatcmpl-{uuid.UUID(int=n)}-{n:08x}"


def exec_id(k):
    return f"execution-{diag._RUN}-{os.getpid()}-{k}"


SCHED_OUTPUTS = []


def scheduler_steps(kind, request_ids, rounds, num_tokens=32, max_tokens=16):
    """Real vLLM scheduling: kind "P" is vLLM's own Scheduler (P runs without a scheduler config, async off in this
    measurement), kind "D" vllm_ascend's AsyncRecomputeScheduler (recompute_scheduler_enable + --async-scheduling).
    All requests are added before the first schedule(); between rounds the output is fed back with one sampled token
    per scheduled request. Returns the SchedulerOutputs."""
    import functools

    from tests.v1.core import utils as core_utils
    from vllm.config import DeviceConfig, VllmConfig
    from vllm.v1.outputs import ModelRunnerOutput

    from vllm_ascend.core import recompute_scheduler as rs

    saved = core_utils.AsyncScheduler, core_utils.VllmConfig
    if kind == "D":
        core_utils.AsyncScheduler = rs.AsyncRecomputeScheduler
    core_utils.VllmConfig = functools.partial(VllmConfig, device_config=DeviceConfig(device="cpu"))
    try:
        scheduler = core_utils.create_scheduler(async_scheduling=(kind == "D"), block_size=16,
                                                max_num_seqs=max(16, len(request_ids)),
                                                max_num_batched_tokens=max(8192, len(request_ids) * num_tokens))
    finally:
        core_utils.AsyncScheduler, core_utils.VllmConfig = saved
    for request_id, request in zip(request_ids, core_utils.create_requests(num_requests=len(request_ids),
                                                                          num_tokens=num_tokens, max_tokens=max_tokens)):
        request.request_id = request_id
        scheduler.add_request(request)
    outputs = []
    for _ in range(rounds):
        output = scheduler.schedule()
        outputs.append(output)
        req_ids = list(output.num_scheduled_tokens.keys())
        scheduler.update_from_output(output, ModelRunnerOutput(
            req_ids=req_ids, req_id_to_index={r: i for i, r in enumerate(req_ids)},
            sampled_token_ids=[[7] for _ in req_ids], logprobs=None, prompt_logprobs_dict={}, pooler_output=[]))
    SCHED_OUTPUTS.append((kind, type(scheduler).__qualname__, [len(o.num_scheduled_tokens) for o in outputs]))
    return outputs


def recompute_scheduler_steps(request_ids, rounds):
    return scheduler_steps("D", request_ids, rounds, max_tokens=rounds)


HOST20 = "d-node-0123456789abc"
RUN32 = "0123456789abcdef0123456789abcdef"
ID_RE = re.compile(r"^([a-z_]+)-([0-9a-f]{32})-(\d+)-(\d+)$")
CLOCK0 = 1_728_000_000_000_000  # 20 days of uptime, 16 digits


def normalize(obj, key=None):
    """Site-like widths, so the sizes do not depend on the container: host 20 characters, pid 1234567, native thread
    id 2345678, run id 32 hex, ids <kind>-<run>-<pid>-<counter> with a 7-digit counter, clock readings (*_ns values of
    10^13 and above) 16 digits."""
    if isinstance(obj, dict):
        return {k: normalize(v, k) for k, v in obj.items()}
    if isinstance(obj, list):
        return [normalize(v, key) for v in obj]
    if isinstance(obj, bool) or obj is None:
        return obj
    if key == "host" and isinstance(obj, str):
        return HOST20
    if key == "pid" and isinstance(obj, int):
        return 1234567
    if key == "native_tid" and isinstance(obj, int):
        return 2345678
    if key == "run_id" and isinstance(obj, str):
        return RUN32
    if isinstance(obj, str):
        m = ID_RE.match(obj)
        if m:
            return f"{m.group(1)}-{RUN32}-1234567-{1000000 + int(m.group(4)) % 1000000:07d}"
        return obj
    if isinstance(obj, int) and key and key.endswith("_ns") and obj >= 10 ** 13:
        return CLOCK0 + obj % 10 ** 14
    return obj


def kind_of(record):
    e = record.get("event")
    for k in ("scope", "phase", "operation", "transfer_role"):
        if record.get(k) is not None:
            e += f"[{k}={record[k]}]"
    return e


PREFIX = 44 + 1  # "INFO 11-19 04:49:47 [pd_diagnostics.py:214] " and the newline (vLLM's formatter, as t51-glm53-p2d)


def sizes(records):
    rows = []
    for r in records:
        n = normalize(r)
        msg = "[PD_DIAG] " + json.dumps(n, separators=(",", ":"))
        rows.append({"kind": kind_of(r), "execution_id": r.get("execution_id"), "request_id": r.get("request_id"),
                     "n_request_ids": len(r["request_ids"]) if isinstance(r.get("request_ids"), list) else None,
                     "n_members": len(r["members"]) if isinstance(r.get("members"), list) else None,
                     "message_bytes": len(msg.encode()), "log_line_bytes": len(msg.encode()) + PREFIX,
                     "normalized": n})
    return rows
