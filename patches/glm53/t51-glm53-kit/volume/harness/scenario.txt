    runner = Runner()
    out = {}
    ids_p1 = [site_request_id(i) for i in range(n)]
    ids_p2 = [site_request_id(100000 + i) for i in range(n)]
    ids_d = [site_request_id(200000 + i) for i in range(n)]
    # P (load_async=false): step 1 schedules n requests, each with a pool hit (sync load) and a save; step 2 releases
    # step 1's requests (wait_for_put) and schedules n new requests of the same kind.
    RECORDS.clear()
    worker = store_connector(load_async=False)
    # each step from vLLM's own Scheduler (P: max_tokens 1, so the requests finish at the step's update)
    p_out1 = scheduler_steps("P", ids_p1, 1, max_tokens=1)[0]
    step(runner, p_out1.pd_execution_id, [request(r, load=True) for r in ids_p1], sync_loads=True, output=p_out1)
    p_out2 = scheduler_steps("P", ids_p2, 1, max_tokens=1)[0]
    step(runner, p_out2.pd_execution_id, [request(r, load=True) for r in ids_p2], released=ids_p1, sync_loads=True,
         output=p_out2)
    drain(worker.kv_send_thread)
    out["P"] = list(RECORDS)
    # D (load_async=true): the first step of n new requests from the real AsyncRecomputeScheduler, each with a pool
    # hit loaded in the background (save per d_save), then KVPoolScheduler._emit_load_links; then decode_steps
    # decode steps of the same n running requests (the scheduler's own next schedule() after update_from_output).
    RECORDS.clear()
    worker = store_connector(load_async=True)
    outputs = recompute_scheduler_steps(ids_d, 1 + decode_steps)
    first = outputs[0]
    meta, output = step(runner, first.pd_execution_id, [request(r, load=True, save=d_save) for r in ids_d],
                        sync_loads=False, output=first)
    drain(worker.kv_send_thread, worker.kv_recv_thread)
    scheduler = object.__new__(KVPoolScheduler)
    scheduler.load_async = True
    scheduler.use_layerwise = False
    scheduler.vllm_config = SimpleNamespace(
        kv_transfer_config=SimpleNamespace(engine_id="1"), parallel_config=SimpleNamespace(data_parallel_rank=0)
    )
    scheduler._emit_load_links(meta, output)
    out["D_first"] = list(RECORDS)
    RECORDS.clear()
    for later in outputs[1:]:
        step(runner, later.pd_execution_id, [], sync_loads=False, output=later)
    drain(worker.kv_send_thread, worker.kv_recv_thread)
    out["D_decode"] = list(RECORDS)
    return out
